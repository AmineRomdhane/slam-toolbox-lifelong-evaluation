cmake_minimum_required(VERSION 3.16)
project(slam_eval_route_controller)
find_package(ament_cmake REQUIRED)
install(PROGRAMS scripts/controller.py DESTINATION lib/${PROJECT_NAME} RENAME route_controller)
install(FILES scripts/control_core.py DESTINATION lib/${PROJECT_NAME})
get_filename_component(BENCHMARK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
install(DIRECTORY "${BENCHMARK_ROOT}/benchmark/routes/" DESTINATION share/${PROJECT_NAME}/routes)
if(BUILD_TESTING)
 enable_testing()
 add_test(NAME control_core COMMAND /usr/bin/python3 ${BENCHMARK_ROOT}/tests/test_controller.py)
endif()
ament_package()
