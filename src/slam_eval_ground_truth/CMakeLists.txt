cmake_minimum_required(VERSION 3.16)
project(slam_eval_ground_truth)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz_math_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-math7 REQUIRED)
add_executable(ground_truth_node src/ground_truth_node.cpp)
target_compile_features(ground_truth_node PRIVATE cxx_std_17)
target_include_directories(ground_truth_node PRIVATE include)
ament_target_dependencies(ground_truth_node rclcpp geometry_msgs)
target_link_libraries(ground_truth_node gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 gz-math7::gz-math7)
target_compile_options(ground_truth_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS ground_truth_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  target_compile_features(core_test PRIVATE cxx_std_17)
  target_include_directories(core_test PRIVATE include)
  target_link_libraries(core_test gz-msgs10::gz-msgs10 gz-math7::gz-math7)
  add_test(NAME core_test COMMAND core_test)
endif()
ament_package()
